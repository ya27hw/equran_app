import 'package:equran/theme/equran_theme.dart';
import 'package:equran/theme/equran_text_styles.dart';
import 'package:equran/l10n/app_localizations.dart';
import 'package:equran/widgets/last_read_cards.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'gallery_test_support.dart';
void main() {
  setUpAll(loadGalleryFonts);
  testWidgets('unchanged empty resume card overflows at 390 width and 1.3 without dock', (tester) async {
    await tester.binding.setSurfaceSize(const Size(390, 844));
    addTearDown(() => tester.binding.setSurfaceSize(null));
    await tester.pumpWidget(MaterialApp(
      theme: EquranTheme.dark(),
      builder: (context, child) {
        final theme = EquranTextStyles.localizeTheme(Theme.of(context), const Locale('en'));
        return Theme(data: theme.copyWith(textTheme: theme.textTheme.apply(fontSizeFactor: .94), primaryTextTheme: theme.primaryTextTheme.apply(fontSizeFactor: .94)), child: child!);
      },
      localizationsDelegates: AppLocalizations.localizationsDelegates,
      supportedLocales: AppLocalizations.supportedLocales,
      home: MediaQuery(data: const MediaQueryData(size: Size(390,844), textScaler: TextScaler.linear(1.3)), child: Scaffold(body: Builder(builder: (context) {
        final l = AppLocalizations.of(context)!;
        return Padding(padding: const EdgeInsets.symmetric(horizontal: 20), child: EquranResumeImageCard(primary: l.beginWithQuran, subtitle: l.startReadingSubtitle, actionText: l.startReading, trailingAssetPath: equranResumeQuranAsset, onTap: () {}));
      }))),
    ));
    final error = tester.takeException();
    expect(error.toString(), contains('overflowed by 3.0 pixels'));
    expect(tester.widgetList(find.byType(NavigationBar)), isEmpty);
  });
}
