import 'dart:io';
import 'package:equran/main.dart' as app;
import 'package:equran/backend/library.dart';
import 'package:equran/backend/startup_coordinator.dart';
import 'package:equran/home/home.dart';
import 'package:equran/home/main_page.dart';
import 'package:equran/home/quran_stats_page.dart';
import 'package:equran/home_dashboard/home_dashboard_page.dart';
import 'package:equran/duas/duas_page.dart';
import 'package:equran/prayer/prayer_times_page.dart';
import 'package:equran/home/more_page.dart';
import 'package:equran/hifz/hifz.dart' as hifz;
import 'package:equran/l10n/app_localizations.dart';
import 'package:equran/prayer/prayer_models.dart';
import 'package:equran/prayer/prayer_settings_store.dart';
import 'package:equran/services/device_capability_profile.dart';
import 'package:equran/services/device_capability_service.dart';
import 'package:equran/widgets/redesign/floating_dock.dart';
import 'package:equran/widgets/redesign/design_icon.dart';
import 'package:equran/widgets/favourites_list.dart';
import 'package:flutter/material.dart';
import 'package:flutter_test/flutter_test.dart';
import 'package:integration_test/integration_test.dart';
import 'package:path_provider/path_provider.dart';

void main() {
  final binding = IntegrationTestWidgetsFlutterBinding.ensureInitialized();
  testWidgets('Round 1 real app at 390px', (tester) async {
    await StartupCoordinator.instance.initializeBlocking();
    await SettingsDB().put('locale', 'en');
    await SettingsDB().put('themeMode', 'dark');
    await SettingsDB().put('themeScheme', 'default');
    await SettingsDB().put('prayerTrackingEnabled', true);
    await SettingsDB().put('quranScriptStyle', 'qpc-hafs');
    await PrayerSettingsStore().saveLocation(
      const PrayerLocation(
        latitude: 21.4225,
        longitude: 39.8262,
        label: 'Makkah',
        mode: PrayerLocationMode.manual,
        countryCode: 'SA',
        timezoneId: 'Asia/Riyadh',
      ),
    );
    await StartupCoordinator.instance.startDeferred();
    final previousErrorHandler = FlutterError.onError;
    var phase = '';
    final baselineOverflows = <String>[];
    FlutterError.onError = (details) {
      final message = details.exceptionAsString();
      if (message.contains(
            'A RenderFlex overflowed by 3.0 pixels on the bottom.',
          ) &&
          phase.contains('1.3')) {
        baselineOverflows.add('$phase: $message');
        debugPrint('KNOWN_BASELINE_OVERFLOW $phase: $message');
        return;
      }
      previousErrorHandler?.call(details);
    };
    addTearDown(() => FlutterError.onError = previousErrorHandler);
    final now = DateTime.now();
    await const QuranBookmarkService().saveBookmarkDetails(
      94,
      5,
      isFavourite: true,
      note: 'With hardship comes ease.',
    );
    for (final verse in [1, 2]) {
      await hifz.HifzDB.saveEntry(
        hifz.HifzEntry()
          ..surah = 1
          ..ayah = verse
          ..status = 'mastered'
          ..dueDate = now.add(const Duration(days: 14))
          ..track = 'manzil',
      );
    }
    final directory = Directory(
      '${(await getApplicationDocumentsDirectory()).path}/round1-captures',
    );
    await directory.create(recursive: true);
    Future<void> capture(String name) async {
      await tester.pump(const Duration(milliseconds: 400));
      final bytes = await binding.takeScreenshot(name);
      await File('${directory.path}/$name.png').writeAsBytes(bytes);
      debugPrint('CAPTURE $name');
    }

    Future<void> tab(int index, Type page) async {
      final dock = find.byType(FloatingDock);
      final buttons = find.descendant(of: dock, matching: find.byType(InkWell));
      await tester.tap(buttons.at(index));
      await tester.pumpAndSettle();
      expect(NavigationBloc.instance.value.selectedIndex, index);
      expect(find.byType(page), findsOneWidget);
      // Reselect uses the same callback and leaves the selected page reachable.
      await tester.tap(buttons.at(index));
      await tester.pumpAndSettle();
      expect(NavigationBloc.instance.value.selectedIndex, index);
      final dockTop = tester.getTopLeft(dock).dy;
      final pageBox = tester.getRect(find.byType(page).first);
      expect(pageBox.bottom, lessThan(dockTop));
    }

    for (final (name, scheme, mode, locale, scale) in [
      ('emerald-dark', 'default', 'dark', 'en', 1.0),
      ('emerald-light', 'default', 'light', 'en', 1.0),
      ('black-dark', 'black', 'dark', 'en', 1.0),
      ('red-dark', 'red', 'dark', 'en', 1.0),
      ('scale-1.3', 'default', 'dark', 'en', 1.3),
      ('arabic', 'default', 'dark', 'ar', 1.0),
      ('arabic-scale-1.3', 'default', 'dark', 'ar', 1.3),
    ]) {
      phase = name;
      await tester.pumpWidget(const SizedBox.shrink());
      await SettingsDB().put('themeScheme', scheme);
      await SettingsDB().put('themeMode', mode);
      await SettingsDB().put('locale', locale);
      tester.platformDispatcher.textScaleFactorTestValue = scale;
      NavigationBloc.instance.selectTab(0);
      await app.main();
      for (var i = 0; i < 30 && find.byType(HomePage).evaluate().isEmpty; i++) {
        await tester.pump(const Duration(milliseconds: 250));
      }
      await tester.pumpAndSettle();
      expect(find.byType(HomePage), findsOneWidget);
      final context = tester.element(find.byType(HomePage));
      expect(MediaQuery.sizeOf(context).width, 390);
      expect(MediaQuery.textScalerOf(context).scale(1), scale);
      expect(
        Directionality.of(context),
        locale == 'ar' ? TextDirection.rtl : TextDirection.ltr,
      );
      final l = AppLocalizations.of(context)!;
      for (final (index, type, suffix) in [
        (0, HomeDashboardPage, 'home'),
        (1, MainPage, 'quran'),
        (2, PrayerTimesPage, 'prayer'),
        (3, DuasPage, 'duas'),
        (4, MorePage, 'more'),
      ]) {
        await tab(index, type);
        await capture('$name-$suffix');
      }
      await tab(1, MainPage);
      await tester.tap(find.text(l.saved));
      await tester.pumpAndSettle();
      await capture('$name-saved');
      final note = find.text(l.note.toUpperCase());
      await tester.ensureVisible(note.first);
      await tester.pumpAndSettle();
      await capture('$name-note');
      // Saved editor is reached through its existing overflow menu.
      await tester.ensureVisible(
        find.byKey(const ValueKey('saved-menu-94-005')),
      );
      await tester.tap(find.byKey(const ValueKey('saved-menu-94-005')));
      await tester.pumpAndSettle();
      await tester.tap(find.text(l.editNote));
      await tester.pumpAndSettle();
      expect(find.byKey(const Key('saved-note-editor')), findsOneWidget);
      await capture('$name-note-editor');
      await tester.tap(find.byTooltip(l.close));
      await tester.pumpAndSettle();
      await tab(4, MorePage);
      // This action is below the fold in the real More page.
      for (
        var i = 0;
        i < 8 && tester.getCenter(find.text(l.statistics)).dy > 620;
        i++
      ) {
        await tester.drag(find.byType(MorePage), const Offset(0, -300));
        await tester.pumpAndSettle();
      }
      expect(tester.getCenter(find.text(l.statistics)).dy, lessThan(620));
      await tester.tap(find.text(l.statistics));
      await tester.pumpAndSettle();
      expect(find.byType(StatisticsPage), findsOneWidget);
      final today = find.byKey(const ValueKey('statistics-today'));
      expect(
        find.descendant(of: today, matching: find.text('2')),
        findsWidgets,
      );
      // Return to the Today card so both its real counts and labels are in view.
      tester
          .state<ScrollableState>(
            find
                .descendant(
                  of: find.byType(StatisticsPage),
                  matching: find.byType(Scrollable),
                )
                .first,
          )
          .position
          .jumpTo(0);
      await tester.pumpAndSettle();
      await capture('$name-statistics');
      await tester.tap(
        find.byTooltip(
          MaterialLocalizations.of(
            tester.element(find.byType(StatisticsPage)),
          ).backButtonTooltip,
        ),
      );
      await tester.pumpAndSettle();
      expect(tester.takeException(), isNull);
    }
    // Native stress captures; no saved order/count is changed.
    final original = NavigationBloc.instance.value;
    for (final count in [4, 5, 6]) {
      final active = [
        NavItem.home,
        NavItem.quran,
        NavItem.prayer,
        if (count >= 5) NavItem.duas,
        if (count == 6) NavItem.readingPlans,
        NavItem.more,
      ];
      NavigationBloc.instance.value = original.copyWith(
        activeNavbarItems: active,
        selectedIndex: 0,
      );
      await tester.pumpAndSettle();
      await capture('dock-$count-arabic-scale-1.3');
      expect(
        find.descendant(
          of: find.byType(FloatingDock),
          matching: find.byType(DesignIcon),
        ),
        findsNWidgets(count),
      );
    }
    NavigationBloc.instance.value = original;
    final service = DeviceCapabilityService.instance;
    final oldProfile = service.value;
    service.value = const DeviceCapabilityProfile(
      lowRam: true,
      processorCount: 2,
    );
    await tester.pumpAndSettle();
    expect(
      find.descendant(
        of: find.byType(FloatingDock),
        matching: find.byType(BackdropFilter),
      ),
      findsNothing,
    );
    await capture('dock-solid-fallback');
    service.value = oldProfile;
    debugPrint('CAPTURE_DIR ${directory.path}');
    binding.reportData!['knownBaselineOverflows'] = baselineOverflows;
    debugPrint('KNOWN_BASELINE_OVERFLOW_COUNT ${baselineOverflows.length}');
    expect(tester.takeException(), isNull);
  }, timeout: const Timeout(Duration(minutes: 15)));
}
